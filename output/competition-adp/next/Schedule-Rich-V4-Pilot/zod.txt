import { z } from 'zod';

const widgetSchema = z.object({
  title: z.string(),
  timeText: z.string(),
  queryId: z.string(),
  dataVersion: z.literal("competition-demo-v1"),
  summary: z.object({
    totalCount: z.number().int().nonnegative(),
    entityType: z.string(),
    entityName: z.string(),
    scope: z.enum(["WEEK", "DAY", "DATE"]),
  }),
  items: z.array(z.object({
    weekday: z.number().int().min(1).max(7),
    date: z.string(),
    periodText: z.string(),
    courseName: z.string(),
    campusName: z.string(),
    building: z.string(),
    roomName: z.string(),
    timeText: z.string(),
    teachers: z.array(z.string()),
    classes: z.array(z.string()),
  })),
  actions: z.array(z.object({
    id: z.string(),
    type: z.literal("sys.chat"),
    label: z.string(),
    query: z.string(),
    intent: z.enum(["schedule_day", "schedule_week", "schedule_choose_day", "schedule_risk_check"]),
    entityType: z.string(),
    entityName: z.string(),
    week: z.number().int().min(1).max(20),
    weekday: z.number().int().min(1).max(7).nullable(),
    date: z.string(),
  })).max(3),
}).strict();

export default widgetSchema;
